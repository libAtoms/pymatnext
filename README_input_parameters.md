# `pymatnext` input TOML parameters

This reference documents every field emitted by the current `pymatnext --defaults`
command. The itemized list mirrors the nesting of the TOML tables and the fields
that can be set in each table. Each field includes its Pydantic type, TOML
default, and description; fields without a default must be specified by the user.
The TOML string `"_NONE_"` is used to set a Python `None` value.

- `[general]` # type: GeneralParams; general parameters
  - `output_filename_prefix = "NS"` # type: str; prefix for all output files
  - `output_filename_prefix_extra = ""` # type: str; extra string to add to output_filename_prefix, designed for easy per-realization overriding
  - `random_seed` *(no default; must be specified by the user)* # type: PositiveInt; seed for random number generator
  - `max_iter = "_NONE_"` # type: PositiveInt | None; maximum NS iteration
  - `stdout_report_interval_s = 60` # type: int; interval in seconds between reports to stdout, < 0 for no reports
  - `sample_interval = 1` # type: int; interval in NS iterations between saved NS samples, <= 0 to disable
  - `traj_interval = 100` # type: int; interval in NS iterations between saved NS configurations, <= 0 to disable
  - `snapshot_interval = 10000` # type: int; interval in NS iterations between restart snapshots, <= 0 to disable
  - `snapshot_save_old = 2` # type: int; how many old snapshots so save
  - `clone_history = false` # type: bool; save a full history of which config was cloned at each iteration
  - `[general.step_size_tune]` # type: StepSizeTuneParams; parameters for tuning step sizes
    - `interval = 1000` # type: PositiveInt; NS iteration interval between step-size tuning
    - `n_configs = 1` # type: PositiveInt; number of configs to run when computing step size related statistics
    - `min_accept_rate = 0.25` # type: PositiveFloat; minimum accept rate for tuning step size
    - `max_accept_rate = 0.5` # type: PositiveFloat; maximum accept rate for tuning step size
    - `adjust_factor = 1.25` # type: PositiveFloat; factor by which step size is multiplied/divided at each tuning iteration
  - `[general.walk_traj_info]` # type: WalkTrajectoryInfoParams; parameters for saving walk trajectories
    - `iter_min = "_NONE_"` # type: NonNegativeInt | None; first iteration at which walk trajectory is saved
    - `iter_max = "_NONE_"` # type: int | None; last iteration (inclusive) at which walk trajectory is saved, negative for no maximum
    - `interval = "_NONE_"` # type: PositiveInt | None; interval at which walk trajectory is saved
    - `avg_times = []` # type: List[PositiveInt] | None; save walk trajectory with time averaging over these time scales
- `[ns]` # type: NSParams; parameters for nested-sampling iteration process
  - `n_walkers` *(no default; must be specified by the user)* # type: int; Number of NS walkers (live points)
  - `walk_length` *(no default; must be specified by the user)* # type: int; Length of NS walk to produce a new, decorrelated config, in number of energy/force evaluations
  - `step_size_tune_walk_length = "_NONE_"` # type: PositiveInt | None; NS walk length used to tune step size
  - `configs_module` *(no default; must be specified by the user)* # type: str; module that defines configurations
  - `initial_config_file = "_NONE_"` # type: str | None; file with initial configurations, if not generated randomly
  - `initial_max_val = "_NONE_"` # type: float | None; value of overriding NS energy initial maxmimum
  - `[ns.exit_conditions]` # type: NSLoopExitParams; parameters for optional exit conditions
    - `module = "_NONE_"` # type: str | None; module that defines an `ExitLoop` class that checks for an exit condition, with `__call__` method that takes current iteration and max value and returns a bool
    - `module_kwargs = {}` # type: Dict[str, Any]; arbitrary kwargs for `ExitLoop` constructor in addition to the `NS` object itself
- `[configs]` # type: ASEAtomsParams; config-type-specific parameters
  - `full_composition = ""` # type: Composition (str | List[str | int]); composition that spans all possible elements that could appear
  - `composition` *(no default; must be specified by the user)* # type: Composition (str | List[str | int]); initial composition of configurations
  - `n_atoms` *(no default; must be specified by the user)* # type: int; number of atoms in each configuration
  - `dims = 3` # type: int; number of dimensions (2 or 3)
  - `pbc = [true, true, true]` # type: List[bool]; periodicity of system along each cell vector
  - `initial_rand_vol_per_atom` *(no default; must be specified by the user)* # type: float; random initial configuration volume per atom
  - `initial_rand_min_dist` *(no default; must be specified by the user)* # type: float; random initial configuration minimum distance between atoms
  - `initial_rand_n_tries = 10` # type: int; number of tries to get initial configuration that obeys minimum distances
  - `[configs.calculator]` # type: CalculatorParams; parameters for calculator of interatomic interactions
    - `type` *(no default; must be specified by the user)* # type: Literal["ASE", "LAMMPS"]; calculator type
    - `args = {}` # type: Dict[str, Any]; arbitrary args for calculator constructor. If 'ASE', 'module' with name of importable module defining a `calc` Calculator object. If 'LAMMPS', 'cmds': list of `pair_style ...` etc. lammps commands, 'types': dict with atomic numbers or chemical symbols as keys and lammps types as values, 'header': optional list of header commands, 'cmd_args': optional command args, 'log_file': optional file for lammps log, 'name': optional lammps shared lib name, 'activate_mliappy_kokkos': optional bool for mliappy kokkos, 'boundary': optional args for lammps boundary command
  - `[configs.walk]` # type: ASEAtomsWalkParams; parameters for NS walks
    - `gmc_traj_len = 8` # type: int; length of GMC walks
    - `cell_traj_len = 8` # type: int; length of cell move walks
    - `type_traj_len = 8` # type: int; length of type (sGC) move walks
    - `gmc_proportion = 0.0` # type: float; proportion of steps to do GMC walks with
    - `cell_proportion = 0.0` # type: float; proportion of steps to do cell walks with
    - `type_proportion = 0.0` # type: float; proportion of steps to do type (sGC) walks with
    - `combined = false` # type: bool; use NS walk function that combines all move types
    - `[configs.walk.max_step_size]` # type: MaxStepSizeParams; maximum step size for each move type
      - `pos_gmc_each_atom = -0.1` # type: float; maximum GMC position move (for each atom), if negative multiplied by cube root of atomic volume
      - `cell_volume_per_atom = -0.05` # type: float; maximum cell volume step (per atom), if negative multiplied by atomic volume
      - `cell_shear_per_rt3_atom = -1.0` # type: float; maximum cell shear step (per natoms^1/3), if negative multiplied by cube root of atomic volume
      - `cell_stretch = 0.2` # type: float; maximum cell stretch (unitless strain)
    - `[configs.walk.step_size]` # type: StepSizeParams; initial step size for each move type
      - `pos_gmc_each_atom = -1.0` # type: float; initial GMC positions move (for each atom)
      - `cell_volume_per_atom = -1.0` # type: float; initial cell volume step (per atom)
      - `cell_shear_per_rt3_atom = -1.0` # type: float; initial cell shear step (per natoms^1/3)
      - `cell_stretch = -1.0` # type: float; initial cell stretch step
    - `[configs.walk.cell]` # type: CellWalkParams; parameters controlling cell walks
      - `min_aspect_ratio = 0.8` # type: float; minimum cell aspect ratio to accept
      - `flat_V_prior = true` # type: bool; use a prior independent of volume, rather than ensemble-correct V^natoms
      - `pressure = 0.0` # type: float; applied pressure in eV/A^3 (or `pressure_GPa` in GPa)
      - `[configs.walk.cell.submove_probabilities]` # type: SubmoveProbabilitiesParams; parameters controlling probabilities of volume, shear, and stretch move types
        - `volume = 0.7` # type: float; probability to try a cell volume move
        - `shear = 0.15` # type: float; probability to try a cell shear move
        - `stretch = 0.15` # type: float; probability to do a cell stretch move
    - `[configs.walk.type]` # type: TypeWalkParams; parameters controlling type (sGC) walks
      - `sGC = false` # type: bool; semi-grand-canonical (species change) moves.
      - `mu = {}` # type: Dict[PositiveInt, float]; dict with atomic numbers as keys and chemical potentials as values for semi-grand-canonical moves
